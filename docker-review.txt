===== BUILD.SH =====
#!/bin/sh
# OPTIONAL production bundle: css/app.min.css and js/app.min.js
# Needs Node.js (npx downloads esbuild the first time).
# The sources are classic scripts that share globals, so they are concatenated
# in index.html order and only minified (names are NOT bundled / renamed).
set -e
cd "$(dirname "$0")"

JS=$(sed -n 's/.*<script defer src="\(js\/[^"]*\)".*/\1/p' index.html | grep -v 'app.min.js')
cat $JS > /tmp/pakoob-all.js
npx --yes esbuild /tmp/pakoob-all.js --minify --outfile=js/app.min.js

cat css/fonts.css css/style.css css/toolbar.css css/responsive.css css/weather.css css/offline.css > /tmp/pakoob-all.css
npx --yes esbuild /tmp/pakoob-all.css --minify --outfile=css/app.min.css

echo "Now in index.html: replace the 16 <script defer src=\"js/...\"> lines with"
echo "  <script defer src=\"js/app.min.js\"></script>"
echo "and the 6 css <link>s with css/app.min.css; update SHELL_FILES in sw.js and bump VERSION."

===== DOWNLOAD-VENDOR.SH =====
#!/bin/sh
# Downloads the third-party libraries into vendor/ (run once, with internet).
set -e
cd "$(dirname "$0")"
# go up until the folder that contains index.html is found
[ -f index.html ] || cd ..

mkdir -p vendor/plotly vendor/leaflet/images vendor/maplibre

curl -fL -o vendor/plotly/plotly-basic-2.35.2.min.js \
  https://cdn.plot.ly/plotly-basic-2.35.2.min.js

L=https://unpkg.com/leaflet@1.9.4/dist
curl -fL -o vendor/leaflet/leaflet.js  $L/leaflet.js
curl -fL -o vendor/leaflet/leaflet.css $L/leaflet.css
for f in layers.png layers-2x.png marker-icon.png marker-icon-2x.png marker-shadow.png; do
  curl -fL -o vendor/leaflet/images/$f $L/images/$f
done

M=https://unpkg.com/maplibre-gl@6.9.0/dist
curl -fL -o vendor/maplibre/maplibre-gl.mjs $M/maplibre-gl.mjs
curl -fL -o vendor/maplibre/maplibre-gl.css $M/maplibre-gl.css

mkdir -p fonts
curl -fL -o fonts/Vazirmatn-variable.woff2 \
  "https://unpkg.com/vazirmatn@33.0.3/fonts/webfonts/Vazirmatn%5Bwght%5D.woff2"

echo "Done. Bump VERSION in sw.js so the new files are cached."

===== SET-DOMAIN.SH =====
#!/bin/sh
# Usage: sh set-domain.sh example.com   (replaces YOUR-DOMAIN.com in robots.txt and sitemap.xml)
set -e
[ -n "$1" ] || { echo "usage: sh set-domain.sh example.com"; exit 1; }
cd "$(dirname "$0")"
sed -i "s/YOUR-DOMAIN\.com/$1/g" robots.txt sitemap.xml index.html
sed -i "s|<lastmod>.*</lastmod>|<lastmod>$(date +%F)</lastmod>|" sitemap.xml
echo "Done: $1"

# Pakoob (پاکوب) — Mountain Route Compare

A browser-based app for analysing and comparing hiking / mountaineering routes
from GPX or KML files. Everything runs in the browser: no backend, no database,
route files are never uploaded. The site uses privacy-friendly analytics (Plausible, no route data); the forecast sends the summit coordinates to Open-Meteo and map tiles come from third-party tile servers.

## Features

- Compare several routes: distance, elevation profile (real / aligned to start /
  aligned to summit / sequential), slope distribution, difficulty score,
  estimated (adjusted Naismith) and recorded ascent time
- 2D map (Leaflet) with speed colouring, 3D terrain map (MapLibre GL) with an
  optional weather overlay (temperature, wind, precipitation, CAPE)
- Live navigation on the route (GPS, deviation alert, pocket mode)
- 7-day / hourly summit forecast from several models (Open-Meteo)
- Rename, re-order (drag) and download routes; routes are kept in IndexedDB
- Offline maps: download the 2D / 3D tiles (and terrain) around the tracks
  in advance (map section -> "دانلود نقشه برای استفادهٔ آفلاین")
- Light / dark mode, Persian RTL interface, installable PWA, works offline
  (app shell, libraries and visited map tiles are cached)

## Adding a default route

Put the GPX in `data/sample/` and add one entry to `data/sample/index.json`
(`routes` for the sidebar list, `quick` for the home-page button). No HTML change.

## Setup / release

1. `sh download-vendor.sh` once (needs internet).
2. Replace the domain: `sh set-domain.sh example.com`.
3. Bump `VERSION` in `sw.js`.

## Technologies

HTML5, CSS3, JavaScript (no framework), Plotly.js, Leaflet, MapLibre GL JS.
